-- ══════════════════════════════════════════════════════════════
--  FLIGHT BOOKING SYSTEM – MS Access SQL
--  Foreign Keys using inline REFERENCES (no CONSTRAINT keyword)
--  Syntax: column_name DATATYPE REFERENCES ParentTable(column)
--  Run each CREATE TABLE ONE AT A TIME in Access SQL View
-- ══════════════════════════════════════════════════════════════


-- TABLE 1: Airlines  (no FK – root table)
CREATE TABLE Airlines (
    airline_id   AUTOINCREMENT PRIMARY KEY,
    airline_name TEXT(100),
    airline_code TEXT(10),
    country      TEXT(50),
    contact      TEXT(15)
);

-- TABLE 2: Airports  (no FK – root table)
CREATE TABLE Airports (
    airport_id   AUTOINCREMENT PRIMARY KEY,
    airport_code TEXT(10),
    airport_name TEXT(100),
    city         TEXT(50),
    country      TEXT(50)
);

-- TABLE 3: Users  (no FK – root table)
CREATE TABLE Users (
    user_id   AUTOINCREMENT PRIMARY KEY,
    full_name TEXT(100),
    email     TEXT(100),
    password  TEXT(50),
    phone     TEXT(15)
);

-- TABLE 4: Flights
-- FK → Airlines, Airports (origin), Airports (destination)
CREATE TABLE Flights (
    flight_id       AUTOINCREMENT PRIMARY KEY,
    flight_number   TEXT(20),
    airline_id      INTEGER REFERENCES Airlines(airline_id),
    origin_id       INTEGER REFERENCES Airports(airport_id),
    destination_id  INTEGER REFERENCES Airports(airport_id),
    departure       TEXT(30),
    arrival         TEXT(30),
    total_seats     INTEGER,
    available_seats INTEGER,
    fstatus         TEXT(20)
);

-- TABLE 5: FlightClasses
-- FK → Flights
CREATE TABLE FlightClasses (
    class_id    AUTOINCREMENT PRIMARY KEY,
    flight_id   INTEGER REFERENCES Flights(flight_id),
    class_name  TEXT(20),
    price       CURRENCY,
    total_seats INTEGER,
    avail_seats INTEGER
);

-- TABLE 6: Seats
-- FK → Flights, FlightClasses
CREATE TABLE Seats (
    seat_id     AUTOINCREMENT PRIMARY KEY,
    flight_id   INTEGER REFERENCES Flights(flight_id),
    class_id    INTEGER REFERENCES FlightClasses(class_id),
    seat_number TEXT(10),
    is_booked   TEXT(5)
);

-- TABLE 7: Bookings
-- FK → Users, Flights
CREATE TABLE Bookings (
    booking_id   AUTOINCREMENT PRIMARY KEY,
    user_id      INTEGER REFERENCES Users(user_id),
    flight_id    INTEGER REFERENCES Flights(flight_id),
    booking_date TEXT(30),
    bstatus      TEXT(20),
    total_amount CURRENCY
);

-- TABLE 8: Passengers
-- FK → Bookings, FlightClasses, Seats
CREATE TABLE Passengers (
    passenger_id AUTOINCREMENT PRIMARY KEY,
    booking_id   INTEGER REFERENCES Bookings(booking_id),
    class_id     INTEGER REFERENCES FlightClasses(class_id),
    seat_id      INTEGER REFERENCES Seats(seat_id),
    full_name    TEXT(100),
    age          INTEGER,
    gender       TEXT(5)
);

-- TABLE 9: Payments
-- FK → Bookings
CREATE TABLE Payments (
    payment_id       AUTOINCREMENT PRIMARY KEY,
    booking_id       INTEGER REFERENCES Bookings(booking_id),
    amount           CURRENCY,
    method           TEXT(20),
    pstatus          TEXT(20),
    payment_type     TEXT(30),
    transaction_date TEXT(30)
);

-- TABLE 10: Cancellations
-- FK → Bookings
CREATE TABLE Cancellations (
    cancellation_id AUTOINCREMENT PRIMARY KEY,
    booking_id      INTEGER REFERENCES Bookings(booking_id),
    cancelled_on    TEXT(30),
    cancelled_by    TEXT(20),
    reason          TEXT(200),
    refund_amount   CURRENCY,
    refund_status   TEXT(20)
);

-- TABLE 11: PriceHistory
-- FK → FlightClasses
CREATE TABLE PriceHistory (
    history_id AUTOINCREMENT PRIMARY KEY,
    class_id   INTEGER REFERENCES FlightClasses(class_id),
    old_price  CURRENCY,
    new_price  CURRENCY,
    changed_on TEXT(30),
    changed_by TEXT(100)
);

-- TABLE 12: ClassUpgrades
-- FK → Passengers, FlightClasses (old), FlightClasses (new)
CREATE TABLE ClassUpgrades (
    upgrade_id       AUTOINCREMENT PRIMARY KEY,
    passenger_id     INTEGER REFERENCES Passengers(passenger_id),
    old_class_id     INTEGER REFERENCES FlightClasses(class_id),
    new_class_id     INTEGER REFERENCES FlightClasses(class_id),
    price_difference CURRENCY,
    upgraded_on      TEXT(30)
);

-- TABLE 13: FlightScheduleChanges
-- FK → Flights
CREATE TABLE FlightScheduleChanges (
    change_id     AUTOINCREMENT PRIMARY KEY,
    flight_id     INTEGER REFERENCES Flights(flight_id),
    old_departure TEXT(30),
    new_departure TEXT(30),
    old_arrival   TEXT(30),
    new_arrival   TEXT(30),
    change_reason TEXT(200),
    changed_on    TEXT(30)
);

-- TABLE 14: Notifications
-- FK → Users, Flights
CREATE TABLE Notifications (
    notification_id AUTOINCREMENT PRIMARY KEY,
    user_id         INTEGER REFERENCES Users(user_id),
    flight_id       INTEGER REFERENCES Flights(flight_id),
    ntype           TEXT(50),
    message         TEXT(255),
    is_read         TEXT(5),
    created_on      TEXT(30)
);
